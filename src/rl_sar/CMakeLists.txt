# Modified 2026-11-11: replace multi-robot/ROS build with WDBLH-only targets.
# Based on rl_sar, Copyright (c) 2024-2025 Ziqi Fan, Apache-2.0.
cmake_minimum_required(VERSION 3.16)
project(wdblh_controller VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "WDBLH serial controller requires Linux")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
get_filename_component(WDBLH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(CTest)

# USE_CMAKE/ENABLE_LIBTORCH retain compatibility with the original QNN command.
option(USE_CMAKE "Standalone WDBLH build (ROS is removed)" ON)
option(ENABLE_LIBTORCH "Removed legacy backend; must remain OFF" OFF)
option(ENABLE_AIDLITE "Enable installed AidLite QNN248/DSP SDK" OFF)
option(ENABLE_ONNX "Enable an explicitly installed ONNX Runtime" OFF)
option(WDBLH_BUILD_CONTROLLER "Compile the controller; never run it during build/tests" ON)
option(WDBLH_BUILD_DIAGNOSTICS "Compile tools which access hardware when run" OFF)
set(ONNX_RUNTIME_DIR "" CACHE PATH "ONNX Runtime package with include/ and lib/")
set(AIDLITE_ROOT "" CACHE PATH "Optional AidLite SDK prefix")
if(NOT USE_CMAKE OR ENABLE_LIBTORCH)
    message(FATAL_ERROR "This extraction supports standalone QNN/optional ONNX only; use USE_CMAKE=ON and ENABLE_LIBTORCH=OFF")
endif()

find_package(Threads REQUIRED)
find_package(TBB REQUIRED)
find_package(yaml-cpp REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
    set(WDBLH_YAML_TARGET yaml-cpp::yaml-cpp)
else()
    set(WDBLH_YAML_TARGET yaml-cpp)
endif()

add_library(wdblh_headers INTERFACE)
target_compile_definitions(wdblh_headers INTERFACE POLICY_DIR="${WDBLH_ROOT}/policy" USE_CMAKE)
target_include_directories(wdblh_headers INTERFACE
    include fsm_robot
    library/core/fsm library/core/inference_runtime library/core/logger
    library/core/loop library/core/observation_buffer library/core/rl_sdk
    library/core/vector_math "${WDBLH_ROOT}/standalone")

add_library(inference_runtime
    library/core/inference_runtime/inference_runtime.cpp
    library/core/inference_runtime/aidlite_model.cpp)
target_link_libraries(inference_runtime PUBLIC wdblh_headers Threads::Threads)
if(ENABLE_AIDLITE)
    find_path(AIDLITE_INCLUDE_DIR aidlux/aidlite/aidlite.hpp
        HINTS "${AIDLITE_ROOT}/include" /usr/local/include)
    find_library(AIDLITE_LIBRARY NAMES aidlite
        HINTS "${AIDLITE_ROOT}/lib" /usr/local/lib)
    if(NOT AIDLITE_INCLUDE_DIR OR NOT AIDLITE_LIBRARY)
        message(FATAL_ERROR "ENABLE_AIDLITE=ON requires installed AidLite headers/library. Set AIDLITE_ROOT or AIDLITE_INCLUDE_DIR/AIDLITE_LIBRARY. QNN248 DSP plugin and device authorization are also required at runtime.")
    endif()
    target_include_directories(inference_runtime PRIVATE "${AIDLITE_INCLUDE_DIR}")
    target_link_libraries(inference_runtime PUBLIC "${AIDLITE_LIBRARY}")
    target_compile_definitions(inference_runtime PRIVATE USE_AIDLITE)
endif()
if(ENABLE_ONNX)
    find_path(ONNX_INCLUDE_DIR onnxruntime_cxx_api.h
        PATHS "${ONNX_RUNTIME_DIR}/include" NO_DEFAULT_PATH)
    find_library(ONNX_LIBRARY NAMES onnxruntime
        PATHS "${ONNX_RUNTIME_DIR}/lib" NO_DEFAULT_PATH)
    if(NOT ONNX_INCLUDE_DIR OR NOT ONNX_LIBRARY)
        message(FATAL_ERROR "ENABLE_ONNX=ON requires ONNX_RUNTIME_DIR with valid headers and library; copied broken symlinks are not usable")
    endif()
    # USE_ONNX changes public class layout: propagate it and its includes.
    target_compile_definitions(inference_runtime PUBLIC USE_ONNX)
    target_include_directories(inference_runtime PUBLIC "${ONNX_INCLUDE_DIR}")
    target_link_libraries(inference_runtime PUBLIC "${ONNX_LIBRARY}")
endif()
if(NOT ENABLE_AIDLITE AND NOT ENABLE_ONNX)
    message(STATUS "Offline build: inference backends disabled; supplied AIDEM cannot run")
endif()

add_library(observation_buffer library/core/observation_buffer/observation_buffer.cpp)
target_link_libraries(observation_buffer PUBLIC wdblh_headers)
add_library(rl_sdk library/core/rl_sdk/rl_sdk.cpp)
target_link_libraries(rl_sdk PUBLIC inference_runtime observation_buffer ${WDBLH_YAML_TARGET} TBB::tbb Threads::Threads)
add_library(pc_protocol "${WDBLH_ROOT}/standalone/pc_protocol.cpp")
target_link_libraries(pc_protocol PUBLIC wdblh_headers)
add_library(dm_imu_raw "${WDBLH_ROOT}/standalone/dm_imu_raw.cpp")
target_link_libraries(dm_imu_raw PUBLIC wdblh_headers Threads::Threads)

if(WDBLH_BUILD_CONTROLLER)
    add_executable(rl_real_wdblh src/rl_real_wdblh.cpp)
    target_link_libraries(rl_real_wdblh PRIVATE rl_sdk pc_protocol dm_imu_raw)
    if(ENABLE_AIDLITE OR ENABLE_ONNX)
        target_compile_definitions(rl_real_wdblh PRIVATE WDBLH_INFERENCE_CONFIGURED)
    endif()
endif()
add_executable(test_inference_runtime "${WDBLH_ROOT}/standalone/test_inference_runtime.cpp")
target_link_libraries(test_inference_runtime PRIVATE inference_runtime)

if(BUILD_TESTING)
    add_executable(test_protocol_dump "${WDBLH_ROOT}/tests/test_protocol_dump.cpp")
    target_link_libraries(test_protocol_dump PRIVATE pc_protocol)
    add_executable(test_wdblh_motor_gate "${WDBLH_ROOT}/standalone/test_wdblh_motor_gate.cpp")
    target_link_libraries(test_wdblh_motor_gate PRIVATE pc_protocol)
    add_test(NAME motor_feedback_gate COMMAND test_wdblh_motor_gate)
    add_executable(test_wdblh_motor_wait_fsm "${WDBLH_ROOT}/standalone/test_wdblh_motor_wait_fsm.cpp")
    target_link_libraries(test_wdblh_motor_wait_fsm PRIVATE rl_sdk)
    add_test(NAME motor_wait_fsm COMMAND test_wdblh_motor_wait_fsm)
    add_executable(test_wdblh_config "${WDBLH_ROOT}/tests/test_wdblh_config.cpp")
    target_link_libraries(test_wdblh_config PRIVATE rl_sdk)
    add_test(NAME policy_config COMMAND test_wdblh_config)
    set_tests_properties(motor_wait_fsm policy_config PROPERTIES
        ENVIRONMENT "WDBLH_POLICY_DIR=${WDBLH_ROOT}/policy")
endif()

if(WDBLH_BUILD_DIAGNOSTICS)
    foreach(tool test_motor_raw test_motor_pos test_motor_debug)
        add_executable(${tool} "${WDBLH_ROOT}/standalone/${tool}.cpp")
        target_link_libraries(${tool} PRIVATE pc_protocol Threads::Threads)
    endforeach()
    foreach(tool test_imu_raw)
        add_executable(${tool} "${WDBLH_ROOT}/standalone/${tool}.cpp")
        target_link_libraries(${tool} PRIVATE dm_imu_raw)
    endforeach()
endif()
